/**
 * FEMOON 予約フォーム バックエンド (GAS Web App)
 * 予約→ ①Sheets書き込み ②Googleカレンダー登録 を同時実行
 * + オーナー専用ダッシュボードAPI (doGet)
 */
const SHEET_NAME = 'FEMOON 予約管理';
const CALENDAR_TZ = 'Asia/Tokyo';
const ADMIN_TOKEN = '653566fda899ea7e9138aca1f67bcc4e';

function getSheet_(){
  const props = PropertiesService.getScriptProperties();
  let id = props.getProperty('SPREADSHEET_ID');
  if(!id){
    const ss = SpreadsheetApp.create(SHEET_NAME);
    id = ss.getId();
    props.setProperty('SPREADSHEET_ID', id);
    const sheet = ss.getActiveSheet();
    sheet.setName('予約');
    sheet.getRange(1,1,1,17).setValues([[
      '予約ID','受付日時','希望日','曜日','時間','コースID','コース名','所要時間','料金',
      'お名前','フリガナ','電話','メール','来店回数','ご要望','状態','カレンダーURL'
    ]]);
    sheet.getRange(1,1,1,17).setFontWeight('bold').setBackground('#fbf0eb');
    sheet.setFrozenRows(1);
    const sm = ss.insertSheet('月次サマリー');
    sm.getRange(1,1,1,4).setValues([['月','予約件数','売上合計','コース内訳']]);
    sm.getRange(1,1,1,4).setFontWeight('bold').setBackground('#fbf0eb');
    sm.setFrozenRows(1);
    sm.getRange(2,1).setFormula('=UNIQUE(TEXT(予約!C2:C,"YYYY-MM"))');
    sm.getRange(2,2).setFormula('=ARRAYFORMULA(IF(LEN(A2:A)=0,,COUNTIF(TEXT(予約!C2:C,"YYYY-MM"),A2:A)))');
    sm.getRange(2,3).setFormula('=ARRAYFORMULA(IF(LEN(A2:A)=0,,SUMIF(TEXT(予約!C2:C,"YYYY-MM"),A2:A,予約!I2:I)))');
    sm.setColumnWidth(4,400);
  }
  return SpreadsheetApp.openById(id).getSheetByName('予約');
}

function addToCalendar_(data, id){
  try{
    const cal = CalendarApp.getDefaultCalendar();
    const [y,m,d] = data.date.split('-').map(Number);
    const [hh,mm] = data.slot.split(':').map(Number);
    const start = new Date(y, m-1, d, hh, mm, 0);
    const end = new Date(start.getTime() + (data.duration||60)*60000);
    const title = '【予約】' + data.course_name + ' / ' + data.name;
    const desc = [
      '予約ID: ' + id,
      'お名前: ' + data.name + (data.kana?(' ('+data.kana+')'):''),
      '電話: ' + data.tel,
      'メール: ' + (data.email||'-'),
      '来店回数: ' + data.visit,
      'コース: ' + data.course_name + ' / ' + data.duration + '分 / ¥' + data.price,
      'ご要望: ' + (data.notes||'-')
    ].join('\n');
    const event = cal.createEvent(title, start, end, {description: desc});
    return event.getId() ? ('https://calendar.google.com/calendar/u/0/r/eventedit/' + Utilities.base64Encode(event.getId()).replace(/=+$/,'')) : '';
  }catch(err){
    return 'ERR: ' + err;
  }
}

function doPost(e){
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try{
    const data = JSON.parse(e.postData.contents);
    const id = 'FM-' + new Date().getTime().toString(36).toUpperCase();
    const calUrl = addToCalendar_(data, id);
    const sheet = getSheet_();
    sheet.appendRow([
      id, new Date(), data.date, data.weekday, data.slot,
      data.course_id, data.course_name, data.duration, data.price,
      data.name, data.kana, data.tel, data.email, data.visit, data.notes, '受付済', calUrl
    ]);
    return ContentService.createTextOutput(JSON.stringify({ok:true,id:id,calendar:calUrl}))
      .setMimeType(ContentService.MimeType.JSON);
  }catch(err){
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:String(err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }finally{
    lock.releaseLock();
  }
}

function doGet(e){
  const params = e.parameter || {};
  const action = params.action || 'status';
  // 認証 (一部のpublicアクションは除外)
  const publicActions = ['status','closed_list_public'];
  if(publicActions.indexOf(action) === -1 && params.token !== ADMIN_TOKEN){
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:'unauthorized'}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if(action === 'status'){
    return ContentService.createTextOutput(JSON.stringify({status:'ok',service:'FEMOON Booking API'}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if(action === 'closed_list_public'){
    return ContentService.createTextOutput(JSON.stringify({ok:true,dates:getClosedDates_()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if(action === 'list'){
    const sheet = getSheet_();
    const rng = sheet.getDataRange().getValues();
    const hdr = rng[0];
    const rows = rng.slice(1).filter(r=>r[0]).map(r=>{
      const obj = {};
      hdr.forEach((h,i)=>obj[h]=r[i]);
      return obj;
    });
    return ContentService.createTextOutput(JSON.stringify({ok:true,rows:rows}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if(action === 'update'){
    const sheet = getSheet_();
    const id = params.id;
    const status = params.status;
    const data = sheet.getDataRange().getValues();
    for(let i=1;i<data.length;i++){
      if(data[i][0]===id){
        sheet.getRange(i+1, 16).setValue(status);
        return ContentService.createTextOutput(JSON.stringify({ok:true,id:id,status:status}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ok:false,error:'not found'}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  // 定休日設定 (個別休業日)
  if(action === 'closed_list'){
    const dates = getClosedDates_();
    return ContentService.createTextOutput(JSON.stringify({ok:true,dates:dates}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if(action === 'closed_add' || action === 'closed_remove'){
    const date = params.date;
    if(!date) return ContentService.createTextOutput(JSON.stringify({ok:false,error:'date required'}))
      .setMimeType(ContentService.MimeType.JSON);
    const list = getClosedDates_();
    let next;
    if(action === 'closed_add') next = [...new Set([...list, date])].sort();
    else next = list.filter(d=>d!==date);
    PropertiesService.getScriptProperties().setProperty('CLOSED_DATES', JSON.stringify(next));
    return ContentService.createTextOutput(JSON.stringify({ok:true,dates:next}))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:false,error:'unknown action'}))
    .setMimeType(ContentService.MimeType.JSON);
}

function getClosedDates_(){
  const raw = PropertiesService.getScriptProperties().getProperty('CLOSED_DATES');
  try{ return raw ? JSON.parse(raw) : []; }catch(e){ return []; }
}

3ウェブアプリとしてデプロイ

右上「デプロイ」→「新しいデプロイ」

  • 種類 → 「ウェブアプリ」を選択
  • 説明 → FEMOON予約API
  • 実行ユーザー → 自分
  • アクセスできるユーザー → 全員

「デプロイ」ボタン押下 → アクセス許可ダイアログが出るので承認。

4発行されたウェブアプリURLを Jobs に貼る

デプロイ完了画面に「ウェブアプリ URL」が表示されます。それをコピーして、チャットに貼って Jobs に送ってください。Jobs がフォームに繋ぎ込みます。
