function setup() {
  const ss = SpreadsheetApp.getActive();
  const data = JSON.parse(UrlFetchApp.fetch(DATA_URL).getContentText());
  // --- list ---
  let sh = ss.getSheetByName(SH_LIST) || ss.insertSheet(SH_LIST);
  sh.clear();
  sh.getRange(1, 1, 1, HEAD.length).setValues([HEAD]);
  const rows = data.targets.map(t => [t[0], t[1], t[2], new Date(t[3] + 'T00:00:00'), t[4], t[5], t[6], t[7], t[8], t[9], t[10], t[11], t[12], t[13], t[14], t[15], '', '未訪問', '', '', '', '', t[16], t[17]]);
  sh.getRange(2, 1, rows.length, HEAD.length).setValues(rows);
  formatList(sh, rows.length);
  // --- route ---
  let rt = ss.getSheetByName(SH_ROUTE) || ss.insertSheet(SH_ROUTE, 0);
  rt.clear();
  const rh = ['ブロック','予定日','担当','エリア','軒数','移動距離(km)','ルート①','ルート②','訪問済','状態'];
  rt.getRange(1, 1, 1, rh.length).setValues([rh]);
  const br = data.blocks.map((b, i) => {
    const r = i + 2;
    return [b[0], new Date(b[1] + 'T00:00:00'), b[2], b[3], b[4], b[5],
      '=HYPERLINK("' + b[6] + '","ルート①")', b[7] ? '=HYPERLINK("' + b[7] + '","ルート②")' : '',
      '=COUNTIFS(\'' + SH_LIST + '\'!B:B,A' + r + ',\'' + SH_LIST + '\'!R:R,"<>未訪問")',
      '=IF(I' + r + '=0,"未着手",IF(I' + r + '>=E' + r + ',"完了","途中"))'];
  });
  rt.getRange(2, 1, br.length, rh.length).setValues(br);
  styleHeader(rt, rh.length); rt.getRange('B2:B').setNumberFormat('m/d(ddd)'); rt.setFrozenRows(1);
  const rules = [
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('完了').setBackground('#93c47d').setRanges([rt.getRange('J2:J')]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('途中').setBackground('#fff2cc').setRanges([rt.getRange('J2:J')]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$B2=TODAY()').setBackground('#cfe2f3').setRanges([rt.getRange('A2:H')]).build()];
  rt.setConditionalFormatRules(rules);
  [8, 11, 6, 12, 6, 11, 10, 10, 8, 10].forEach((w, i) => rt.setColumnWidth(i + 1, w * 8));
  // --- excluded ---
  let ex = ss.getSheetByName(SH_EX) || ss.insertSheet(SH_EX);
  ex.clear();
  const eh = ['店名','ジャンル','住所','電話','評価','口コミ数','HP','理由'];
  ex.getRange(1, 1, 1, eh.length).setValues([eh]);
  if (data.excluded.length) ex.getRange(2, 1, data.excluded.length, eh.length).setValues(data.excluded);
  styleHeader(ex, eh.length); ex.setFrozenRows(1);
  // --- log ---
  let lg = ss.getSheetByName(SH_LOG) || ss.insertSheet(SH_LOG);
  if (lg.getLastRow() === 0) { lg.getRange(1, 1, 1, 8).setValues([['日時','担当','ID','店名','結果','次にやること','フォロー日','メモ']]); styleHeader(lg, 8); lg.setFrozenRows(1); }
  // --- summary ---
  let sm = ss.getSheetByName(SH_SUM) || ss.insertSheet(SH_SUM);
  sm.clear();
  sm.getRange(1, 1, 1, 4).setValues([['結果', '合計', '担当A', '担当B']]); styleHeader(sm, 4);
  RESULTS.forEach((r, i) => {
    const row = i + 2;
    sm.getRange(row, 1, 1, 4).setValues([[r[0], '=COUNTIF(\'' + SH_LIST + '\'!R:R,A' + row + ')', '=COUNTIFS(\'' + SH_LIST + '\'!R:R,A' + row + ',\'' + SH_LIST + '\'!E:E,"A")', '=COUNTIFS(\'' + SH_LIST + '\'!R:R,A' + row + ',\'' + SH_LIST + '\'!E:E,"B")']]);
    sm.getRange(row, 1).setBackground(r[1]);
  });
  const tr = RESULTS.length + 2;
  sm.getRange(tr, 1, 1, 4).setValues([['合計', '=SUM(B2:B' + (tr - 1) + ')', '=SUM(C2:C' + (tr - 1) + ')', '=SUM(D2:D' + (tr - 1) + ')']]).setFontWeight('bold');
  sm.getRange(tr + 2, 1, 2, 2).setValues([['今日までにフォローが必要', '=COUNTIFS(\'' + SH_LIST + '\'!T:T,"<="&TODAY(),\'' + SH_LIST + '\'!R:R,"興味あり・要フォロー")'], ['成約率（訪問済のうち）', '=IFERROR(B7/(B' + tr + '-B2),0)']]);
  sm.getRange(tr + 3, 2).setNumberFormat('0.0%');
  sm.setColumnWidth(1, 200);
  try {
    const chart = sm.newChart().setChartType(Charts.ChartType.BAR).addRange(sm.getRange('A2:B' + (tr - 1)))
      .setPosition(1, 6, 0, 0).setOption('title', '訪問結果').setOption('legend', { position: 'none' }).build();

  } catch (e) {}
  const def = ss.getSheetByName('シート1'); if (def && ss.getSheets().length > 1) ss.deleteSheet(def);
  ss.setActiveSheet(rt);
}

function styleHeader(sh, n) {
  sh.getRange(1, 1, 1, n).setBackground('#2c7f8e').setFontColor('#ffffff').setFontWeight('bold');
}

function formatList(sh, n) {
  styleHeader(sh, HEAD.length);
  sh.setFrozenRows(1); sh.setFrozenColumns(6);
  sh.getRange(2, C['予定日'], n).setNumberFormat('m/d(ddd)');
  sh.getRange(2, C['フォロー日'], n, 2).setNumberFormat('m/d');
  const resRange = sh.getRange(2, C['結果'], n);
  resRange.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(RESULTS.map(r => r[0]), true).build());
  sh.getRange(2, C['次にやること'], n).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(NEXTS.filter(x => x), true).build());
  sh.getRange(2, C['フォロー日'], n, 2).setDataValidation(SpreadsheetApp.newDataValidation().requireDate().build());
  const all = sh.getRange(2, 1, Math.max(n, 2000), HEAD.length);
  const rules = RESULTS.filter(r => r[0] !== '未訪問').map(r =>
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$R2="' + r[0] + '"').setBackground(r[1]).setRanges([all]).build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=AND($T2<>"",$T2<=TODAY(),$R2="興味あり・要フォロー")').setBackground('#f6b26b').setFontColor('#000000').setRanges([sh.getRange(2, C['フォロー日'], Math.max(n, 2000))]).build());
  sh.setConditionalFormatRules(rules);
  const widths = { 'ID': 40, 'ブロック': 56, '順番': 44, '予定日': 76, '担当': 40, '店名': 200, 'ジャンル': 96, '種別': 110, '町名': 80, '住所': 260, '電話': 110, '評価': 44, '口コミ数': 56, 'HP状況': 90, 'SNS・ポータル': 180, 'Googleマップ': 70, '試作HP': 90, '結果': 130, '次にやること': 120, 'フォロー日': 76, '訪問日': 76, 'メモ': 260, '緯度': 60, '経度': 60 };
  HEAD.forEach((h, i) => sh.setColumnWidth(i + 1, widths[h] || 80));
  sh.hideColumns(C['ID']); sh.hideColumns(C['緯度'], 2);
  const links = sh.getRange(2, C['Googleマップ'], n).getValues().map(r => [r[0] ? '=HYPERLINK("' + r[0] + '","地図")' : '']);
  sh.getRange(2, C['Googleマップ'], n).setValues(links);
  sh.getRange(1, 1, n + 1, HEAD.length).createFilter();
}

/** 追加取り込み：DATA_URL の targets のうち、まだ無いIDだけ末尾に足す */
function importNew() {
  const ss = SpreadsheetApp.getActive(), sh = ss.getSheetByName(SH_LIST);
  const data = JSON.parse(UrlFetchApp.fetch(DATA_URL).getContentText());
  const have = new Set(sh.getRange(2, 1, Math.max(sh.getLastRow() - 1, 1), 1).getValues().map(r => r[0]));
  const add = data.targets.filter(t => !have.has(t[0])).map(t => [t[0], t[1], t[2], new Date(t[3] + 'T00:00:00'), t[4], t[5], t[6], t[7], t[8], t[9], t[10], t[11], t[12], t[13], t[14], t[15] ? '=HYPERLINK("' + t[15] + '","地図")' : '', '', '未訪問', '', '', '', '', t[16], t[17]]);
  if (add.length) sh.getRange(sh.getLastRow() + 1, 1, add.length, HEAD.length).setValues(add);
  SpreadsheetApp.getActive().toast(add.length + '件 追加しました');
}

/** 夜間の試作HP登録など: POST {key, protos:[{id,url}]} */
function doPost(e) {
  const body = JSON.parse(e.postData.contents || '{}');
  if (body.key !== API_KEY) return ContentService.createTextOutput('ng');
  const sh = SpreadsheetApp.getActive().getSheetByName(SH_LIST);
  const ids = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(r => r[0]);
  let n = 0;
  (body.protos || []).forEach(p => {
    const i = ids.indexOf(p.id);
    if (i >= 0) { sh.getRange(i + 2, C['試作HP']).setFormula('=HYPERLINK("' + p.url + '","試作HP")'); n++; }
  });
  return ContentService.createTextOutput(JSON.stringify({ ok: true, updated: n })).setMimeType(ContentService.MimeType.JSON);
}

function fmtDate_(d) { return d instanceof Date ? Utilities.formatDate(d, 'Asia/Tokyo', 'yyyy-MM-dd') : (d || ''); }

function readRows_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(SH_LIST);
  const n = sh.getLastRow() - 1;
  const vals = sh.getRange(2, 1, n, HEAD.length).getValues();
  const forms = sh.getRange(2, C['Googleマップ'], n, 2).getFormulas();
  const pick = f => { const m = /HYPERLINK\("([^"]+)"/.exec(f || ''); return m ? m[1] : ''; };
  return vals.map((v, i) => ({
    row: i + 2, id: v[0], block: v[1], seq: v[2], date: fmtDate_(v[3]), who: v[4], name: v[5], genre: v[6], cat: v[7], town: v[8], addr: v[9], tel: v[10],
    rating: v[11], reviews: v[12], hp: v[13], sns: v[14], map: pick(forms[i][0]), proto: pick(forms[i][1]), result: v[17], next: v[18], follow: fmtDate_(v[19]), visited: fmtDate_(v[20]), memo: v[21], lat: v[22], lng: v[23]
  }));
}

function api_day(who, date) {
  const rows = readRows_();
  let mine = rows.filter(r => r.who === who && r.date === date);
  const blocks = {};
  const rt = SpreadsheetApp.getActive().getSheetByName(SH_ROUTE);
  const bv = rt.getRange(2, 1, rt.getLastRow() - 1, 8).getFormulas();
  const bvals = rt.getRange(2, 1, rt.getLastRow() - 1, 8).getValues();
  bvals.forEach((b, i) => { blocks[b[0]] = { no: b[0], date: fmtDate_(b[1]), who: b[2], area: b[3], km: b[5], links: [bv[i][6], bv[i][7]].map(f => { const m = /HYPERLINK\("([^"]+)"/.exec(f || ''); return m ? m[1] : ''; }).filter(x => x) }; });
  const bnos = [...new Set(mine.map(r => r.block))];
  // 予定日の無い日は、自分の担当で未完了の一番早いブロックを提案
  let suggested = false;
  if (!mine.length) {
    const open = rows.filter(r => r.who === who && r.result === '未訪問').sort((a, b) => a.date < b.date ? -1 : a.date > b.date ? 1 : a.seq - b.seq);
    if (open.length) { const b = open[0].block; mine = rows.filter(r => r.block === b); bnos.push(b); suggested = true; }
  }
  mine.sort((a, b) => a.block - b.block || a.seq - b.seq);
  return { rows: mine, blocks: bnos.map(n => blocks[n]).filter(Boolean), suggested };
}

function api_follow() {
  const today = fmtDate_(new Date());
  return readRows_().filter(r => ['興味あり・要フォロー', '話せた・検討中', '商談の約束'].includes(r.result))
    .sort((a, b) => (a.follow || '9999') < (b.follow || '9999') ? -1 : 1).map(r => Object.assign(r, { due: r.follow && r.follow <= today }));
}

function api_summary() {
  const rows = readRows_(), out = {};
  RESULTS.forEach(r => out[r[0]] = { all: 0, A: 0, B: 0, color: r[1] });
  rows.forEach(r => { const o = out[r.result] || (out[r.result] = { all: 0, A: 0, B: 0, color: '#fff' }); o.all++; if (r.who === 'A' || r.who === 'B') o[r.who]++; });
  return { total: rows.length, byResult: out };
}

function api_save(p) {
  const lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    const ss = SpreadsheetApp.getActive(), sh = ss.getSheetByName(SH_LIST);
    const row = Number(p.row);
    if (sh.getRange(row, 1).getValue() !== p.id) throw new Error('行がずれています。読み込み直してください');
    sh.getRange(row, C['結果']).setValue(p.result);
    sh.getRange(row, C['次にやること']).setValue(p.next || '');
    sh.getRange(row, C['フォロー日']).setValue(p.follow ? new Date(p.follow + 'T00:00:00') : '');
    sh.getRange(row, C['訪問日']).setValue(new Date());
    if (p.memo != null) sh.getRange(row, C['メモ']).setValue(p.memo);
    ss.getSheetByName(SH_LOG).appendRow([new Date(), p.who || '', p.id, sh.getRange(row, C['店名']).getValue(), p.result, p.next || '', p.follow || '', p.memo || '']);
    return { ok: true };
  } finally { lock.releaseLock(); }
}

function api_log(id) {
  const lg = SpreadsheetApp.getActive().getSheetByName(SH_LOG);
  if (lg.getLastRow() < 2) return [];
  return lg.getRange(2, 1, lg.getLastRow() - 1, 8).getValues().filter(r => r[2] === id)
    .map(r => ({ at: Utilities.formatDate(r[0], 'Asia/Tokyo', 'M/d HH:mm'), who: r[1], result: r[4], next: r[5], memo: r[7] }));
}
