rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ============ Rooms / Buildings (公開マスタ) ============
    match /rooms/{roomId} {
      allow read: if true;
      allow write: if false;
    }
    match /buildings/{bldgId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ Reservations ============
    // LIFF からは認証無しで予約作成可能 (デモ運用)
    // - 必須フィールドあり + 妥当な金額 + status=pending のみ受付
    match /reservations/{resNo} {
      allow read: if true;
      allow create: if request.resource.data.lineUserId is string
        && request.resource.data.name is string
        && request.resource.data.tel is string
        && request.resource.data.totalPrice is number
        && request.resource.data.totalPrice > 0
        && request.resource.data.status == 'pending';
      // 更新は admin 画面から (デモは open update を許可・本番は管理画面側で IP 制限等を別途)
      allow update: if true;
      allow delete: if false;
    }

    // ============ Guests ============
    match /guests/{lineUserId} {
      allow read, write: if true;
    }

    // ============ Availability (空室カレンダー) ============
    match /availability/{date}/rooms/{roomId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ Ops Logs ============
    match /ops_logs/{logId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ Ops State ============
    match /ops_state/{docId} {
      allow read: if true;
      allow write: if false;
    }
  }
}
