rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ============ 公開マスタ (read only) ============
    match /rooms/{roomId} {
      allow read: if true;
      allow write: if false;
    }
    match /buildings/{bldgId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 空室カレンダー (LIFF が public read) ============
    // 個人情報なし (status / heldBy のみ) なので read 開放
    match /availability/{date}/rooms/{roomId} {
      allow read: if true;
      allow write: if false;  // Functions admin SDK 経由のみ
    }

    // ============ 予約データ (Functions のみ書き換え可) ============
    match /reservations/{resNo} {
      allow read: if false;
      allow create: if false;
      allow update: if false;
      allow delete: if false;
    }

    // ============ ゲスト (個人情報、Functions のみ) ============
    match /guests/{lineUserId} {
      allow read, write: if false;
    }

    // ============ 運用ログ (admin api 経由) ============
    match /ops_logs/{logId} {
      allow read, write: if false;
    }

    // ============ オーナー設定 (admin api 経由) ============
    match /ops_state/{docId} {
      allow read, write: if false;
    }

    // ============ OTA 設定 (admin api 経由) ============
    match /ota_config/{source} {
      allow read, write: if false;
    }

    // ============ admin セッション (Functions のみ) ============
    match /admin_sessions/{token} {
      allow read, write: if false;
    }
  }
}
